cmake_minimum_required(VERSION 3.16)
project(goose_archive_native LANGUAGES C)
find_package(LibArchive 3.6 REQUIRED)
find_package(ZLIB 1.2.11 REQUIRED)
add_library(goose_archive_native STATIC native/archive.c)
set_target_properties(goose_archive_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(goose_archive_native PRIVATE c_std_11)
target_compile_options(goose_archive_native PRIVATE -Wall -Wextra -Werror -Wframe-larger-than=16384)
target_link_libraries(goose_archive_native PUBLIC LibArchive::LibArchive ZLIB::ZLIB)
